# Pinned Python exploitation toolkit for the lab image (Listing 2-1 / Figure 2-1).
#
# These versions are the ones the book was written and tested against. They are
# pinned EXACTLY so that a fresh `./scripts/lab.sh --build` today reproduces the
# same toolchain the chapter exploits were authored for. Unpinned installs pull
# whatever is newest on PyPI, and newer pwntools / ROPgadget / ropper releases
# change APIs and output formats, which is what made chapter tests drift and
# fail (see issue #1).
#
# CI (.github/workflows/ci.yml) installs from this same file, so CI and the lab
# always agree.
#
# To intentionally move the toolchain forward: bump a version here, rebuild,
# run `make doctor` and `make test-all`, and update Figure 2-1 to match.
pwntools==4.12.0
ROPgadget==7.4
ropper==1.13.8
capstone==5.0.1
unicorn==2.0.1.post1
keystone-engine==0.9.2
pyelftools==0.31
# pytest is used by the ch10/ch28 test harnesses. Pin it in the image so those
# chapters don't pip-install it at test time — an unpinned runtime install needs
# network access and hangs on a constrained/offline host (the ch10 check-tools
# stall). CI installs this same file, so CI and the lab stay in lockstep.
pytest==8.3.4
